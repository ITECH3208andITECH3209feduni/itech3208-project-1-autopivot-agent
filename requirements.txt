# AutoPivot — core dependencies (no machine learning)
# Developed by Vadim Rudoi
#
# Enough to run authentication, the dashboard and the database tooling:
#
#     uvicorn api.app:app --reload
#
# Every package here has a prebuilt wheel on Apple Silicon and installs in
# seconds. For the vehicle processing pipeline, use requirements-ml.txt instead.

# Server
fastapi
uvicorn[standard]
python-multipart
python-dotenv

# Auth
bcrypt>=4.0,<5.0
PyJWT>=2.8,<3.0
email-validator

# Image handling — validating and measuring uploads needs no ML
pillow

# Backdrop geometry (backdrop_analysis.py). A dealer's backdrop is measured when
# it is uploaded — where its floor begins and where its horizon falls — and
# upload is a light-half route, so the geometry has to work here. These are
# ordinary numerical wheels rather than a model stack: they install in seconds,
# pull in no GPU runtime and no torch, so the split that requirements-ml.txt
# draws is unchanged. Without them a backdrop could only be measured on a
# machine with a GPU, which is the wrong shape entirely for an upload form.
numpy
opencv-python-headless

# Listing-URL import (api/url_import.py). Core rather than ML: fetching a page
# and reading its <img> tags needs neither torch nor a GPU, and both install
# in seconds.
beautifulsoup4
httpx

# Database
SQLAlchemy>=2.0,<3.0
alembic>=1.13,<2.0
psycopg[binary]>=3.1,<4.0
